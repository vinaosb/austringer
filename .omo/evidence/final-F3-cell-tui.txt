fn plain(code: crossterm::event::KeyCode) -> Key {
    Key {
        code,
        mods: crossterm::event::KeyModifiers::NONE,
    }
}

fn ctrl(c: char) -> UiEvent {
    UiEvent::Key(Key {
        code: crossterm::event::KeyCode::Char(c),
        mods: crossterm::event::KeyModifiers::CONTROL,
    })
}

fn type_and_submit(app: &mut App, text: &str) -> Vec<UiAction> {
    let mut actions = Vec::new();
    actions.extend(app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Char('i')))));
    for c in text.chars() {
        actions.extend(app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Char(c)))));
    }
    actions.extend(app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Enter))));
    actions
}

async fn dispatch(wired: &Wired, actions: Vec<UiAction>) {
    for action in actions {
        let cmd = match action {
            UiAction::Send(msg) => ClientCmd::Send(msg),
            UiAction::Reconnect => ClientCmd::Reconnect,
            UiAction::Shutdown => ClientCmd::Shutdown,
        };
        let _ = wired.cmd.send(cmd).await;
    }
}

async fn serve(state: Arc<DaemonState>) -> String {
    let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.expect("bind");
    let port = listener.local_addr().expect("addr").port();
    let app = build_router(Arc::clone(&state));
    tokio::spawn(async move {
        let serve = axum::serve(
            listener,
            app.into_make_service_with_connect_info::<std::net::SocketAddr>(),
        );
        let _ = serve.await;
    });
    format!("127.0.0.1:{port}")
}

async fn attach(base: &str, session: &str, token: &str) -> (App, Wired) {
    let wired = wire(base, session, token);
    let mut app = App::new(DEFAULT_CONTEXT_WINDOW);
    assert!(
        wait_until(
            &mut app,
            &wired,
            |a| matches!(a.conn, Conn::Attached),
            Duration::from_secs(5)
        ),
        "attach; conn {:?}",
        app.conn
    );
    (app, wired)
}

/// All user-visible transcript/notice lines (entries + client notes).
fn surface_join(app: &App) -> String {
    use harness_tui::transcript::Entry;
    let mut lines: Vec<String> = app
        .view
        .entries()
        .iter()
        .map(|e| match e {
            Entry::User(t) | Entry::Assistant(t) | Entry::System(t) | Entry::Notice(t)
            | Entry::Condensed(t) => t.clone(),
        })
        .collect();
    lines.extend(app.view.notes().iter().cloned());
    lines.join("\n")
}

enum Step {
    Turn(AssistantTurn),
    Fail(String),
}

struct ScriptStack {
    steps: std::sync::Mutex<std::collections::VecDeque<Step>>,
}

#[async_trait]
impl Inference for ScriptStack {
    async fn infer(
        &mut self,
        _request: &InferenceRequest<'_>,
        _cancel: &CancelToken,
    ) -> Result<AssistantTurn, InferenceError> {
        match self.steps.lock().expect("steps").pop_front() {
            Some(Step::Turn(turn)) => Ok(turn),
            Some(Step::Fail(reason)) => Err(InferenceError::Failure(reason)),
            None => Err(InferenceError::Failure("script exhausted".to_owned())),
        }
    }
}

fn call(id: &str, tool: &str, arguments: serde_json::Value) -> PendingCall {
    PendingCall {
        call_id: id.into(),
        tool: tool.into(),
        arguments,
    }
}

fn text_turn(text: &str, calls: Vec<PendingCall>) -> AssistantTurn {
    AssistantTurn {
        blocks: vec![ContentBlock::Text { text: text.to_owned() }],
        tool_calls: calls,
        usage: None,
    }
}

/// The per-model provider-stack source for routed daemons (the
/// dispatcher-live pattern): hands each model its scripted stack once.
fn script_source(
    scripts: Vec<(&str, Vec<Step>)>,
) -> ModelSource {
    let map: std::collections::HashMap<String, Vec<Step>> = scripts
        .into_iter()
        .map(|(model, steps)| (model.to_owned(), steps))
        .collect();
    let map = std::sync::Mutex::new(map);
    Arc::new(move |model: &str| {
        let mut guard = map.lock().expect("scripts");
        guard.remove(model).map(|steps| {
            Box::new(ScriptStack {
                steps: std::sync::Mutex::new(steps.into()),
            }) as Box<dyn Inference>
        })
    })
}

// routing.json: the session `default` role on grok-4-7 with a
// one-hop chain onto grok-4-6. Both price through the builtin xai
// table's grok-4 alias (version-boundary prefix match), so an
// acknowledged hop is visible in the HUD cost row with real dollars.
const ROUTING: &str = r#"
{
  "provider": "xai",
  "roles": [ { "role": "default", "model": "grok-4-7" } ],
  "fallbacks": [ { "model": "grok-4-7", "chain": ["grok-4-6"] } ],
  "rate_limit_wait_out_secs": 1,
  "retry_attempts": 1
}
"#;

/// A routed daemon over the scripted per-model stacks (the configured
/// identity is openai/gpt-6-sol â€” distinct from every routed model so
/// a mis-route would be unambiguous; the routed path replaces the
/// factory stack entirely).
fn routed_daemon(
    ask_timeout: Duration,
    scripts: Vec<(&str, Vec<Step>)>,
) -> (Arc<DaemonState>, tempfile::TempDir) {
    let dir = tempfile::tempdir().expect("tempdir");
    let factory: harness_daemon::usage::InferenceFactory = Arc::new(|| ComposedInference {
        inference: Box::new(NeverServes),
        identity: Some(ModelIdentity::new("openai", "gpt-6-sol")),
    });
    let mut state = DaemonState::new(base_config(&dir, ask_timeout), factory).expect("state");
    let table = RoutingTable::from_json(ROUTING, &Catalog::seeded()).expect("routing doc");
    state.routing = Some(Arc::new(RoutingIntake {
        table,
        models: script_source(scripts),
    }));
    (Arc::new(state), dir)
}

struct NeverServes;

#[async_trait]
impl Inference for NeverServes {
    async fn infer(
        &mut self,
        _request: &InferenceRequest<'_>,
        _cancel: &CancelToken,
    ) -> Result<AssistantTurn, InferenceError> {
        Err(InferenceError::Failure(
            "factory stack must never serve on the routed path".to_owned(),
        ))
    }
}

fn outage() -> Step {
    Step::Fail("provider unreachable: connection refused".to_owned())
}

    );

}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn cell2b_fallback_deny_keeps_and_timeout_retries_same_model() {
    const DEGRADED: &str =
        "model fallback asked; no interactive client â€” retrying same model";
    let (state, _dir) = routed_daemon(
        Duration::from_millis(200),
        vec![
            ("grok-4-7", vec![outage(), outage(), outage()]),
            ("grok-4-6", vec![Step::Turn(priced_turn("never"))]),
        ],
    );
    let base = serve(state).await;
    let (mut app, wired) = attach(&base, "f3-cell2b", "f3-cell-token").await;

    dispatch(&wired, type_and_submit(&mut app, "deny then ignore")).await;
    // Ask 1: answer n â€” the answer pins the CURRENT identity.
    assert!(
        wait_until(
            &mut app,
            &wired,
            |a| a.fallback_gate().is_some(),
            Duration::from_secs(10)
        ),
        "first ask; surface:\n{}",
        surface_join(&app)
    );
    let deny = app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Char('n'))));
    match deny.as_slice() {
        [UiAction::Send(harness_daemon::protocol::ClientMsg::FallbackAnswer { chosen, .. })] => {
            assert_eq!(
                chosen.model, "grok-4-7",
                "n keeps the current model (never a silent hop)"
            );
        }
        other => panic!("expected a fallback answer, got {other:?}"),
    }
    dispatch(&wired, deny).await;
    // Ask 2: leave it UNACKNOWLEDGED â€” it times out (200 ms), the
    // degraded-mode notice publishes verbatim, the modal reopens once
    // and then the turn fails typed instead of looping.
    assert!(
        wait_until(
            &mut app,
            &wired,
            |a| surface_join(a).matches(DEGRADED).count() >= 1,
            Duration::from_secs(10)
        ),
        "verbatim degraded retry notice; surface:\n{}",
        surface_join(&app)
    );
    assert!(
        wait_until(
            &mut app,
            &wired,
            |a| a.view.turns().last().is_some_and(|t| t.outcome.is_some()),
            Duration::from_secs(15)
        ),
        "turn finished after unacknowledged retries; surface:\n{}",
        surface_join(&app)
    );
    assert!(
        wait_until(
            &mut app,
            &wired,
            |a| surface_join(a).matches("resolved: next model grok-4-7").count() >= 2,
            Duration::from_secs(5)
        ),
        "both resolutions kept the CURRENT model; surface:\n{}",
        surface_join(&app)
    );
    let surface = surface_join(&app);
    println!("=== cell2b surface ===\n{surface}");
    let degraded = surface.matches(DEGRADED).count();
    println!("=== cell2b degraded notices: {degraded}");
    assert!(degraded >= 2, "retry notice repeated per unacknowledged ask: {surface}");
    assert!(
        !surface.contains("next model grok-4-6"),
        "no resolution ever named the hop: {surface}"
    );
    // ZERO substitution events: the hop model has no Usage row.
    assert!(
        !app.hud.cost_line().contains("grok-4-6"),
        "hop never served: {}",
        app.hud.cost_line()
    );
    let _ = wired.cmd.send(ClientCmd::Shutdown).await;
}

// ---------------------------------------------------------------------------
// Mechanism 3: cost governance â€” Usage frames past a tiny cap: HUD cost
// updates, CostWarning at 80%, CostStop at 100%, further submits typed
// refused naming the cap. Uses a PRICED identity (claude-sonnet-4-5)
// with per-turn usage of 25,920 micro-USD against a 30,000 cap.
// ---------------------------------------------------------------------------
/// Cyclic script: edit_lines call (priced usage), then task_completed
/// (unpriced) â€” repeats for every turn.
struct CycleEditInference {
    edit_args: serde_json::Value,
    first: std::sync::atomic::AtomicBool,
}

#[async_trait]
impl Inference for CycleEditInference {
    async fn infer(
        &mut self,
        _request: &InferenceRequest<'_>,
        _cancel: &CancelToken,
    ) -> Result<AssistantTurn, InferenceError> {
        let first = self
            .first
            .swap(false, std::sync::atomic::Ordering::SeqCst);
        Ok(if first {
            AssistantTurn {
                blocks: vec![ContentBlock::Text {
                    text: "editing now".into(),
                }],
                tool_calls: vec![call("t1", "edit_lines", self.edit_args.clone())],
                usage: Some(harness_core::agent_loop::inference::TurnUsage {
                    input_tokens: 7_600,
                    output_tokens: 80,
                    cache_read_tokens: 6_400,
                    cache_write_tokens: 0,
                }),
            }
        } else {
            self.first
                .store(true, std::sync::atomic::Ordering::SeqCst);
            AssistantTurn {
                blocks: vec![ContentBlock::Text {
                    text: "done: turn completed".into(),
                }],
                tool_calls: Vec::new(),
                usage: None,
            }
        })
    }
}

                    router.deliver(
                        &call_id,
                        true,

                            "text": "edited by mock worker",

                                "path": "src/demo.rs",
                                "before_hash": "",
                                "after_hash": "0123456789ab",

    }
    println!("=== cell3 hub after turn 1 ===\n{}", hub_seen.join("\n"));
    println!(

        app.hud.cost_line(),
        app.hud.cache_line()
    );
    assert!(

        "daemon published cost_warning(25920/30000)"

    assert!(

        "HUD cost row live: {}",
        app.hud.cost_line()
    );
    assert_eq!(app.hud.cache_line(), "cache: 84.2%");

    // u128 wire-decode probe (the prior F3 cell found these frames
    // undecodable; serde 1.0.229 should round-trip them now).

        br#"{"type":"event","frame":{"id":99,"event":{"type":"cost_warning","consumed_micros":25920,"cap_micros":30000}}}"#,
    );
    println!("=== cell3 u128 decode probe: {probe:?}");

    println!("=== cell3 hub after turn 2 ===\n{}", hub_seen.join("\n"));

            "=== cell3 COST-FRAME DECODE BUG still present: u128 frames dropped by the client"

    assert!(diff_screen.contains("+new line"), "hunk body rendered: {diff_screen}");

    assert_eq!(app.view.pending_hunks(), 4);
    let accept = app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Char('y'))));

    app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Char('y'))));
    app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Char('n'))));
    assert_eq!(app.view.pending_hunks(), 0);
    let decided = screen(&app, 110, 34);

    assert!(decided.contains("[rejected]"), "decision renders: {decided}");
    let surface = surface_join(&app);
    println!("=== cell3 hunk notes ===\n{surface}");
    assert!(surface.contains("hunk accepted: src/demo.rs"));
    assert!(surface.contains("hunk rejected: src/demo.rs"));
    app.step(ctrl('5'));

    let _ = wired.cmd.send(ClientCmd::Shutdown).await;
}

#[test]
fn cell7_scroll_dispatch_moves_window_and_refollows() {
    let mut app = App::new(DEFAULT_CONTEXT_WINDOW);
    for i in 0..40u64 {
        app.step(UiEvent::Server(user_frame(i, &format!("line-{i:02}"))));
    }
    app.step(ctrl('1'));
    assert_eq!(app.scroll_offset(), 0, "auto-follow at the tail");
    let tail = screen(&app, 80, 24);
    assert!(tail.contains("you: line-39"), "tail:\n{tail}");
    for _ in 0..5 {
        app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Up)));
    }
    println!("=== cell7 scroll_offset after 5x Up: {}", app.scroll_offset());
    assert!(app.scroll_offset() > 0, "window lifted off the tail");
    let scrolled = screen(&app, 80, 24);
    println!("=== cell7 scrolled screen ===\n{scrolled}");
    assert!(scrolled.contains("you: line-18"), "scrolled:\n{scrolled}");
    assert!(!scrolled.contains("you: line-39"), "tail left: {scrolled}");
    for _ in 0..5 {
        app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Down)));
    }
    assert_eq!(app.scroll_offset(), 0, "back at the tail");
    let back = screen(&app, 80, 24);
    assert!(back.contains("you: line-39"), "back:\n{back}");
    for _ in 0..3 {
        app.step(UiEvent::Key(plain(crossterm::event::KeyCode::Up)));
    }
    assert!(app.scroll_offset() > 0);
    app.step(UiEvent::Server(user_frame(40, "line-40")));
    assert_eq!(app.scroll_offset(), 0, "new frame refollows");
    let followed = screen(&app, 80, 24);
    println!("=== cell7 refollow screen ===\n{followed}");
    assert!(followed.contains("you: line-40"), "followed:\n{followed}");
}

## U128 wire fix re-proof

**Defect (F3 TUI QA cell):** `SessionEvent::CostWarning`/`CostStop` carried
`consumed_micros`/`cap_micros` as u128 on the wire. serde's internally-tagged
enum deserialization buffers variant content into `Content` first, and
`Content` cannot represent u128 - so every client's `decode_server` rejected
the frames with `Err(Malformed("u128 is not supported"))` and the TUI silently
dropped cost events: the cost Warning/Stop badge never surfaced.

**Fix:** wire fields of the cost event variants are u64 (micro-USD fits u64
comfortably: max ~18 trillion USD). harness-core `MicroUsd`/budget internals
stay u128; the conversion happens once at the event-construction boundary
(`cost_ledger.rs` `wire_micros`: `u64::try_from(..).unwrap_or(u64::MAX)`,
documented saturation - a spend past u64::MAX micros is a budget that already
exploded and the event still fires). compat.rs v1 downgrade, TUI transcript,
CLI render consume sites are type-agnostic (literals/format) - unchanged except
the new decode test mirrors. Warning at 80% / Stop at 100% semantics untouched.

**RED proof (before fix):** new test
`protocol::tests::cost_event_frames_decode_through_internally_tagged_content`
decoded a real `cost_warning`/`cost_stop` wire frame through `decode_server`
and failed exactly as the cell observed::
